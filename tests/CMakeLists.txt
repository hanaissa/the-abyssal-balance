function(add_placeholder_test name)
    add_executable(${name} ${name}/main.c)
    target_link_libraries(${name} PRIVATE abyssal_core)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/src)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
add_placeholder_test(RFID_Test)
add_placeholder_test(WaterLevel_Test)
add_placeholder_test(Potentiometer_Test)
add_placeholder_test(Pump_Test)
add_placeholder_test(LCDButtons_Test)
add_placeholder_test(CameraOCR_Test)
add_placeholder_test(Integration_Test)
