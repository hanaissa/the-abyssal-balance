function(add_placeholder_test name)
    add_executable(${name} ${name}/main.c)
    target_link_libraries(${name} PRIVATE abyssal_core)
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/src)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
if(ABYSSAL_USE_PICO_SDK)
    add_executable(RFID_Test RFID_Test/main.c)
    target_include_directories(RFID_Test PRIVATE ${CMAKE_SOURCE_DIR}/drivers/rfid)
    target_link_libraries(RFID_Test PRIVATE abyssal_core)
    pico_enable_stdio_usb(RFID_Test 1)
    pico_enable_stdio_uart(RFID_Test 0)
    pico_add_extra_outputs(RFID_Test)
endif()
