cmake_minimum_required(VERSION 3.13)
option(ABYSSAL_PICOCAMERA_HOST_TESTS "Build frame-contract tests without Pico SDK" OFF)
if(ABYSSAL_PICOCAMERA_HOST_TESTS)
    project(PicoCameraContract C CXX)
    add_executable(PicoCameraContract_Test test_frame.c)
    target_include_directories(PicoCameraContract_Test PRIVATE
        ../../../drivers/camera/vendor/PicoCamera/src ../../../include)
    enable_testing()
    add_test(NAME PicoCameraContract_Test COMMAND PicoCameraContract_Test)
    add_executable(PicoCameraCapture_Test test_capture.cpp)
    target_compile_features(PicoCameraCapture_Test PRIVATE cxx_std_17)
    target_include_directories(PicoCameraCapture_Test PRIVATE mock_sdk)
    if(MSVC)
        target_compile_options(PicoCameraCapture_Test PRIVATE /UNDEBUG)
    else()
        target_compile_options(PicoCameraCapture_Test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME PicoCameraCapture_Test COMMAND PicoCameraCapture_Test)
    add_executable(CameraDriverHost_Test test_driver_host.c ../../../drivers/camera/Camera.c)
    target_include_directories(CameraDriverHost_Test PRIVATE ../../../drivers/camera ../../../include)
    add_test(NAME CameraDriverHost_Test COMMAND CameraDriverHost_Test)
    return()
endif()
message(FATAL_ERROR "Use -DABYSSAL_PICOCAMERA_HOST_TESTS=ON for regressions, or tools/camera for the maintained hardware preview.")
