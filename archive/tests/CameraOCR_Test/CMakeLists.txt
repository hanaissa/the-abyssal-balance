cmake_minimum_required(VERSION 3.13)
option(ABYSSAL_CAPTURE_HOST_TESTS "Run camera geometry software checks without hardware" OFF)
if(ABYSSAL_CAPTURE_HOST_TESTS)
    project(AbyssalCaptureGeometry C)
    add_executable(CaptureGeometry_Test test_frame_geometry.c)
    enable_testing()
    add_test(NAME CaptureGeometry_Test COMMAND CaptureGeometry_Test)
    return()
endif()
set(PICO_BOARD pico CACHE STRING "Camera diagnostic board")
include(../../../pico_sdk_import.cmake)
project(AbyssalCameraDiagnostic C CXX ASM)
pico_sdk_init()

add_executable(Camera_Diagnostic diagnostic_main.c)
target_compile_features(Camera_Diagnostic PRIVATE c_std_11)
target_include_directories(Camera_Diagnostic PRIVATE ../../../include)
target_link_libraries(Camera_Diagnostic PRIVATE pico_stdlib hardware_clocks hardware_i2c hardware_pwm)
pico_enable_stdio_usb(Camera_Diagnostic 1)
pico_enable_stdio_uart(Camera_Diagnostic 0)
pico_add_extra_outputs(Camera_Diagnostic)

add_executable(Camera_Capture capture.c ../../../drivers/uv/UV_LED.c)
target_compile_features(Camera_Capture PRIVATE c_std_11)
target_include_directories(Camera_Capture PRIVATE ../../../include)
target_link_libraries(Camera_Capture PRIVATE pico_stdlib hardware_clocks
    hardware_i2c hardware_pwm hardware_pio hardware_dma)
pico_generate_pio_header(Camera_Capture ${CMAKE_CURRENT_LIST_DIR}/camera_capture.pio)
pico_enable_stdio_usb(Camera_Capture 1)
pico_enable_stdio_uart(Camera_Capture 0)
pico_add_extra_outputs(Camera_Capture)
